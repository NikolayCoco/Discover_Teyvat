				text = AG_Pay_30_tooltip_2
				set_variable = {
					name = AG_Primogem_Blessing
					value = 0
				}
			}
		}
	}

	}

AG_Convert_to_Archonism = {
	type = internationalorganization
	show_in_gui_list = no

	automation_tick = never
	automation_tick_frequency = 12
	
	potential = {
		scope:actor = { NOT = { religion.group = religion_group:Archonism } }
	}

	allow = {
		scope:actor = {
			var:AG_Primogem >= 1600
			any_cabinet_character = {
				OR = {
					DvT_is_Anemo_Archon = yes
					DvT_is_Geo_Archon = yes
					DvT_is_Electro_Archon = yes
					DvT_is_Dendro_Archon = yes
					DvT_is_Hydro_Archon = yes
					DvT_is_Hydro_Sovereign = yes
					DvT_is_Pyro_Archon = yes
					DvT_is_Luno_Godness = yes
					DvT_is_Cryo_Archon = yes
					DvT_is_Other_Archon = yes
				}
				modifier:is_head_of_cabinet = yes
			}
		}
	}

	select_trigger = {
		looking_for_a = religion
		target_flag = target_religion
		name = "convert_religion_select_religion"
		none_available_msg_key = "convert_religion_no_religions"
		column = {
			data = name
		}
		column = {
			data = religion_percentage_in_country
		}
		default_sort = religion_percentage_in_country
		visible = {
			group = religion_group:Archonism
			scope:actor = {
				any_cabinet_character = {
					OR = {
						DvT_is_Anemo_Archon = yes
						DvT_is_Geo_Archon = yes
						DvT_is_Electro_Archon = yes
						DvT_is_Dendro_Archon = yes
						DvT_is_Hydro_Archon = yes
						DvT_is_Hydro_Sovereign = yes
						DvT_is_Pyro_Archon = yes
						DvT_is_Luno_Godness = yes
						DvT_is_Cryo_Archon = yes
						DvT_is_Other_Archon = yes
					}
					modifier:is_head_of_cabinet = yes
					religion = root
				}
			}
		}
	}

	effect = {
		scope:actor = {
			add_stability = -100
			AG_Pay_1600 = yes
			custom_tooltip = {
				text = AG_Convert_to_Archonism_tooltip
				change_religion = scope:target_religion
				random_owned_location = {
					limit = {
						is_capital = yes
					}
					every_pop = {
						change_pop_religion = scope:target_religion
					}
				}
			}
		}
	}

	ai_will_do = {
		value = 0
		
	}
}
