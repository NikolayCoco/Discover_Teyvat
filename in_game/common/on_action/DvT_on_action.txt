on_game_start = {
	on_actions = {
		on_DvT_start
	}
}
on_DvT_start = {
    effect = {
        # 1.4：不再依赖 main_menu/setup/<日期> 里的 setup 文件来创建「冒险家协会」。
        # 新开档时由脚本直接创建这个唯一的 IO；如果 setup 已经创建过（exists = yes）就跳过，避免重复创建。
        if = {
            limit = { NOT = { exists = international_organization:DvT_adventurer_guild_IO } }
            every_country = {
                limit = { is_ai = no }
                if = {
                    limit = { NOT = { exists = international_organization:DvT_adventurer_guild_IO } }
                    create_international_organization = {
                        type = international_organization_type:DvT_adventurer_guild_IO
                    }
                }
            }
        }
        if = {
            limit = { NOT = { has_global_variable = DvT_2D_portrait_enabled } }
            set_global_variable = DvT_2D_portrait_enabled
        }
        if = {
            limit = { NOT = { has_global_variable = DvT_start_end } }
            set_global_variable = DvT_start_end
            every_country = { trigger_event_non_silently = { id = DvT_events.1 } }
        }
    }
}
on_character_death = {
    on_actions = {
		on_DvT_character_death
	}
}
on_DvT_character_death = {
    trigger = {
        is_target_in_global_variable_list = {
            name = DvT_characters_created_list
            target = scope:target
        }
	}
    effect = {
        remove_list_global_variable = {
            name = DvT_characters_created_list
            target = scope:target
        }
    }
}
on_annex = {
    on_actions = {
		on_DvT_annex
	}
}
on_DvT_annex = {
	effect = {
		if = {
			limit = {
				root = {
					is_member_of_international_organization = international_organization:DvT_adventurer_guild_IO
				}
				scope:target = {
                    any_character = { DvT_2D_portrait_trigger = yes }
				}
			}
			scope:target = {
                every_character = {
                    limit = {
                        DvT_2D_portrait_trigger = yes
                    }
                    move_country = root
                }
      		}			
		}
    }
}
