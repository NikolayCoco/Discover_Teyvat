DvT_CI_can_marry = {
    message = no                   # Show notification
    on_own_nation = yes             # Can be used within own nation

    select_trigger = {
		looking_for_a = character
		source = actor
		target_flag = recipient
	}

    potential = {
        # Conditions for interaction to appear
        scope:recipient = {
            OR = {
                has_character_modifier = DvT_character_base_modifier
                has_character_modifier = DvT_character_base_modifier_immortal
            }
        }
    }

    allow = {
        # Conditions to enable interaction
    }

    effect = {
        # What happens when interaction is used
        scope:recipient = {
            if = {
                limit = { has_character_modifier = DvT_character_base_modifier }
                remove_character_modifier = DvT_character_base_modifier
                add_character_modifier = {
                    modifier = DvT_character_base_modifier_marriage
                    years = -1
                    mode = replace
                }
            }
            if = {
                limit = { has_character_modifier = DvT_character_base_modifier_immortal }
                remove_character_modifier = DvT_character_base_modifier_immortal
                add_character_modifier = {
                    modifier = DvT_character_base_modifier_marriage_immortal
                    years = -1
                    mode = replace
                }
            }
        }
    }

    ai_tick = never
    ai_will_do = {
		value = 0
		
	}
}

DvT_create_Genshin_Land = {
    message = yes                   # Show notification
    on_own_nation = yes             # Can be used within own nation

    select_trigger = {
		looking_for_a = character
		source = actor
		target_flag = recipient
	}

    potential = {
        # Conditions for interaction to appear
        scope:recipient = {
            DvT_2D_portrait_trigger = yes
        }
    }

    allow = {
        # Conditions to enable interaction
        scope:actor = {
            has_estate_privilege = estate_privilege:vision_estate_Genshin_Land
            var:AG_Primogem >=1600
			or = {
				is_subject = no
				subject_type = {
					allow_subject_creation = yes
				}
			}
			custom_tooltip = {
				text = create_province_subject_union_tt
				trigger_if = {
					limit = { exists = union }
					union ?= {
						OR = {
							NOT = { has_active_senior_partner = yes }
							country_is_senior_partner = { country = scope:actor }
						}
					}
				}
			}
		}
        scope:recipient = {
            can_become_a_ruler = yes
        }
    }

    select_trigger = {
		looking_for_a = province
		interaction_source_list = {
			scope:actor = {
				every_province = {
					add_to_list = source
				}
			}
		}
		target_flag = target_province
		name = "create_province_subject_select_province"
		none_available_msg_key = "create_province_subject_no_province"
		column = {
			data = name
		}
		visible = {
		}
	}

    effect = {
        # What happens when interaction is used
        scope:actor = {
            AG_Pay_1600 = yes
        }
        scope:target_province = {
            create_location_country_from_province = {
                ruler = scope:recipient
                subject_type = subject_type:DvT_genshin_subject

                capital = {
                    province = {
                        save_scope_as = new_province
                    }
                }

                save_scope_as = new_Genshin_Land
            }
        }
    }

    ai_tick = never
    ai_will_do = {
		value = 0
		
	}
}

DvT_CI_Subject_Ruler = {
    message = yes                   # Show notification
    on_own_nation = yes             # Can be used within own nation

    # 第一步：选择要派遣的原神角色
    select_trigger = {
        looking_for_a = character
        source = actor
        target_flag = recipient
    }

    potential = {
        scope:recipient = {
            DvT_2D_portrait_trigger = yes
        }
    }

    allow = {
        scope:actor = {
            has_estate_privilege = estate_privilege:vision_estate_Genshin_Land
        }
    }

    # 第二步：选择要派遣到的属国（任意类型）
    select_trigger = {
        looking_for_a = country
        interaction_source_list = {
            scope:actor = {
                every_subject = {
                    add_to_list = source
                }
            }
        }
        target_flag = target
        name = "choose_country"
        column = {
            data = name
        }
        visible = {
            is_subject_of = scope:actor
        }
        enabled = {
            # 原神领免费；其他属国需 1600 原石
            trigger_if = {
                limit = {
                    is_subject_type = DvT_genshin_subject
                }
                always = yes
            }
            trigger_else = {
                scope:actor = {
                    var:AG_Primogem >= 1600
                }
            }
        }
    }

    effect = {
        scope:actor = {
            if = {
                limit = {
                    scope:target = {
                        NOT = { is_subject_type = DvT_genshin_subject }
                    }
                }
                AG_Pay_1600 = yes
            }
        }
        if = {
            limit = {
                exists = scope:recipient
                exists = scope:target
            }
            scope:recipient = {
                move_country = scope:target
            }
            scope:target = {
                set_new_ruler = scope:recipient
            }
        }
    }

    ai_tick = never
    ai_will_do = {
        value = 0
    }
}
