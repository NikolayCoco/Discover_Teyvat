DvT_create_character = {
	create_character = {
        first_name = $nameF$
        last_name = $nameL$
        religion = religion:$religion$
        culture = culture:$culture$
        age = $age$
        # birth_date = $birth_day$
        # birth_location = location:$birth_place$
        estate = estate_type:vision_estate
        adm = { value = { 0 5 } add = $ability_level$ }
        dip = { value = { 0 5 } add = $ability_level$ }
        mil = { value = { 0 5 } add = $ability_level$ }
        female = $female_bool$
        if = {
            limit = { $immortal$ = yes }
            add_character_modifier = {
                modifier = DvT_character_base_modifier_immortal
                years = -1
                mode = add_and_extend
            }
        }
        else = {
            add_character_modifier = {
                modifier = DvT_character_base_modifier
                years = -1
                mode = add_and_extend
            }
        }
        remove_traits_of_category = ruler
        add_trait = trait:DvT_$code$_01_trait
        add_trait = trait:DvT_$code$_01_trait_leader
        add_trait = trait:DvT_$element$_trait
        add_trait = trait:DvT_$belong$_trait
        save_scope_as = target_character
    }
    random_owned_location = {
        limit = {
            is_capital = yes
        }
        add_pop = {
            type = pop_type:vision_pop
            size = 0.001
            religion = scope:actor.religion
            culture = culture:$culture$
        }
    }
    trigger_event_non_silently = { id = DvT_character_events.1 }
    add_to_global_variable_list = {
        name = DvT_characters_created_list
        target = scope:target_character
    }
}
DvT_CI_Outfit_effect = {
    if = {
        limit = {
            has_trait = DvT_$code$_01_trait_leader
        }
        if = {
            limit = {
                has_trait = DvT_$code$_01_trait
            }
            remove_trait = trait:DvT_$code$_01_trait
            add_trait = trait:DvT_$code$_02_trait
        }
        else = {
            limit = {
                has_trait = DvT_$code$_02_trait
            }
            remove_trait = trait:DvT_$code$_02_trait
            add_trait = trait:DvT_$code$_01_trait
        }
    }
}
