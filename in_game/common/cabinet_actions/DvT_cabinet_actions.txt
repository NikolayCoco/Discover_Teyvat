DvT_Convert_pop_to_Archonism = {
	ability = adm
	allow_multiple = no
	icon = promote_religion
	years = 1
	potential = {
		religion.group = religion_group:Archonism
		num_locations > 0
		any_cabinet_character = {
			OR = {
				DvT_is_Anemo_Archon = yes
				DvT_is_Geo_Archon = yes
				DvT_is_Electro_Archon = yes
				DvT_is_Dendro_Archon = yes
				DvT_is_Hydro_Archon = yes
				DvT_is_Hydro_Sovereign = yes
				DvT_is_Pyro_Archon = yes
				DvT_is_Luno_Godness = yes
				DvT_is_Cryo_Archon = yes
			}
			modifier:is_head_of_cabinet = yes
		}
	}

	allow = {
		any_cabinet_character = {
			OR = {
				DvT_is_Anemo_Archon = yes
				DvT_is_Geo_Archon = yes
				DvT_is_Electro_Archon = yes
				DvT_is_Dendro_Archon = yes
				DvT_is_Hydro_Archon = yes
				DvT_is_Hydro_Sovereign = yes
				DvT_is_Pyro_Archon = yes
				DvT_is_Luno_Godness = yes
				DvT_is_Cryo_Archon = yes
			}
			modifier:is_head_of_cabinet = yes
		}
		religious_unity < 1.0
		var:AG_Primogem >= 8000
	}

	select_trigger = {
		looking_for_a = religion
		target_flag = target_religion
		name = "promote_religion_select_religion"
		none_available_msg_key = "convert_religion_no_religions"
		column = {
			data = name
		}
		column = {
			data = religion_percentage_in_country
		}
		default_sort = religion_percentage_in_country
		visible = {
			group = religion_group:Archonism
			scope:actor.religion = root
		}
	}
	
	select_trigger = {
		looking_for_a = province
		source = actor
		target_flag = selected_province
		
		
		name = "promote_religion_select_province"
		none_available_msg_key = "no_valid_provinces"
		pre_evaluation_sort_value = {
			value = province_tax_base
		}
		pre_evaluation_number_to_evaluate_fully = 5
		column = {
			data = name
		}
		column = {
			data = control
		}
		column = {
			data = population
		}
		column = {
			data = province_tax_base
		}
		visible = {
			religion_percentage = {
				religion = scope:actor.religion
				value < 1 
			}
		}
		map_color = {
			lerp = {
				min_color = define:NMapColors|MAP_COLOR_LOW
				max_color = define:NMapColors|MAP_COLOR_HIGH
				factor = "province.religion_percentage(scope:target_religion)"
			}
		}
	}

	on_activate = {
		
	}

	on_fully_activated = {
		custom_tooltip = {
			text = DvT_pay_8000_Convert_pop_text
			scope:actor = {
				AG_Pay_1600 = yes
				AG_Pay_1600 = yes
				AG_Pay_1600 = yes
				AG_Pay_1600 = yes
				AG_Pay_1600 = yes
			}
			scope:selected_province = {
				every_location_in_province = {
					every_pop = {
						change_pop_religion = scope:target_religion
					}
				}
			}
		}
	}

	progress = {
		cabinet_action:DvT_Convert_pop_to_Archonism = {
			value = "implementation_progress_percentage(scope:actor)"
		}
	}

	is_finished = {
		cabinet_action:DvT_Convert_pop_to_Archonism = {
			"implementation_progress_percentage(scope:actor)" = 1.0
		}
	}

	map_marker = {
		show_for_owner = yes
	}
}
